package com.knightlight.game; import android.util.Log; import java.util.logging.Logger; import java.io.Serializable; public class Battle implements Serializable { private static final long serialVersionUID = 1L; private static final String TAG = "Battle"; public enum State { PLAYER_TURN, ENEMY_TURN, VICTORY, DEFEAT, FLED } public enum Action { ATTACK, USE_POTION, RUN, DEFEND } private Player player; private Enemy enemy; private CombatSystem combatSystem; private State state; private String lastMessage; private final transient Object stateLock = new Object(); /** * Initialize a battle between player and enemy. * CRITICAL: Both player and enemy MUST be non-null. */ public Battle(Player player, Enemy enemy) { if (player == null) { throw new IllegalArgumentException("Player cannot be null"); } if (enemy == null) { throw new IllegalArgumentException("Enemy cannot be null"); } this.player = player; this.enemy = enemy; this.combatSystem = new CombatSystem(player, enemy); this.state = State.PLAYER_TURN; this.lastMessage = "Battle started! " + enemy.getName() + " appears!"; Logger.getLogger(getClass().getName()).info(TAG + ": Battle initialized: Player (Lvl " + player.getLevel() + ") vs " + enemy.getName() + " (Health: " + enemy.getHealth() + ")"); } public Player getPlayer() { return player; } public Enemy getEnemy() { return enemy; } public State getState() { synchronized (stateLock) { return state; } } public String getLastMessage() { return lastMessage; } public CombatSystem getCombatSystem() { return combatSystem; } /** * Process a player action. Must be called when state is PLAYER_TURN. * Transitions to ENEMY_TURN or terminal state (VICTORY/DEFEAT/FLED). */ public void playerAction(Action action) { if (action == null) { Log.w(TAG, "playerAction called with null action"); return; } synchronized (stateLock) { if (state != State.PLAYER_TURN) { Log.w(TAG, "playerAction called when state is " + state); return; } switch (action) { case ATTACK: combatSystem.playerAttack(); lastMessage = combatSystem.getLastLogEntry(); Log.d(TAG, "Player attack: " + lastMessage); break; case USE_POTION: if (player.usePotion()) { lastMessage = "You healed yourself!"; Log.d(TAG, "Player used potion. Health: " + player.getCurrentHealth()); } else { lastMessage = "No potions left!"; Log.w(TAG, "Player attempted potion use but none available"); return; // Action failed, stay in PLAYER_TURN } break; case RUN: if (combatSystem.attemptFlee()) { state = State.FLED; lastMessage = "You fled successfully!"; Log.i(TAG, "FLED: Player escaped from " + enemy.getName()); return; // Exit — battle is over (fled) } else { lastMessage = "Failed to flee!"; Log.d(TAG, "Player failed flee attempt"); // Fall through to enemyTurn() — enemy counterattacks } break; case DEFEND: lastMessage = "You brace for impact!"; Log.d(TAG, "Player defended"); break; default: Log.w(TAG, "Unknown action: " + action); return; } // Check victory condition AFTER player action if (!enemy.isAlive()) { state = State.VICTORY; lastMessage = "You defeated " + enemy.getName() + "!"; Logger.getLogger(getClass().getName()).info(TAG + ": VICTORY: Player defeated " + enemy.getName() + " | Player Health: " + player.getCurrentHealth()); // endCombat() already processes rewards internally — DO NOT DUPLICATE CombatResult result = combatSystem.endCombat(); if (result == null) { Log.e(TAG, "CRITICAL: endCombat() returned null — rewards may not have been processed!"); } else if (!result.isSuccessful()) { Log.e(TAG, "endCombat() failed: " + result.getMessage()); } else { Logger.getLogger(getClass().getName()).info(TAG + ": Rewards processed: XP=" + result.getExperienceAwarded() + " Gold=" + result.getGoldAwarded()); } return; // Battle over — VICTORY } // Transition to enemy turn state = State.ENEMY_TURN; enemyTurn(); } } /** * Execute enemy's turn. Must be called when state is ENEMY_TURN. * Transitions to PLAYER_TURN or DEFEAT. */ public void enemyTurn() { synchronized (stateLock) { if (state != State.ENEMY_TURN) { Log.w(TAG, "enemyTurn called when state is " + state); return; } combatSystem.enemyAttack(); lastMessage = combatSystem.getLastLogEntry(); Log.d(TAG, "Enemy turn: " + lastMessage); // Check defeat condition AFTER enemy attack if (!player.isAlive()) { state = State.DEFEAT; lastMessage = enemy.getName() + " defeated you!"; Logger.getLogger(getClass().getName()).info(TAG + ": DEFEAT: Player defeated by " + enemy.getName() + " | Enemy Health: " + enemy.getHealth()); CombatResult result = combatSystem.endCombat(); if (result == null) { Log.e(TAG, "CRITICAL: endCombat() returned null on defeat"); } else { Log.i(TAG, "Defeat processed: " + result.getMessage()); } return; // Battle over — DEFEAT } // Return to player turn state = State.PLAYER_TURN; } } /** * Called from game loop if needed.
