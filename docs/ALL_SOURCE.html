package com.knightlight.game;

import android.util.Log;
import java.io.Serializable;

/**
 * Battle state manager with thread-safe state access.
 * Uses actual Player/Enemy APIs from Character inheritance chain.
 * FIXED: Proper dungeon reference for enemy cleanup on victory.
 */
public class Battle implements Serializable {
    private static final long serialVersionUID = 2L;
    private static final String TAG = "Battle";

    public enum State {
        PLAYER_TURN, ENEMY_TURN, VICTORY, DEFEAT, FLED
    }

    public enum Action {
        ATTACK, USE_POTION, RUN, DEFEND
    }

    private Player player;
    private Enemy enemy;
    private transient CombatSystem combatSystem;
    private transient Dungeon dungeon;
    private State state;
    private String lastMessage;
    private final transient Object stateLock = new Object();

    public Battle(Player player, Enemy enemy) {
        this(player, enemy, null);
    }

    public Battle(Player player, Enemy enemy, Dungeon dungeon) {
        if (player == null) {
            throw new IllegalArgumentException("Player cannot be null");
        }
        if (enemy == null) {
            throw new IllegalArgumentException("Enemy cannot be null");
        }
        this.player = player;
        this.enemy = enemy;
        this.dungeon = dungeon;
        this.combatSystem = new CombatSystem(player, enemy);
        this.state = State.PLAYER_TURN;
        this.lastMessage = "Battle started! " + enemy.getName() + " appears!";
        Log.d(TAG, "Battle init: Player Lvl " + player.getLevel()
            + " vs " + enemy.getName() + " HP:" + enemy.getCurrentHealth());
    }

    public Object readResolve() {
        if (player != null && enemy != null) {
            this.combatSystem = new CombatSystem(player, enemy);
        } else {
            Log.w(TAG, "readResolve: null player/enemy, marking battle ended");
            this.state = State.VICTORY;
        }
        return this;
    }

    public Player getPlayer() { return player; }
    public Enemy getEnemy() { return enemy; }
    public Dungeon getDungeon() { return dungeon; }
    public void setDungeon(Dungeon dungeon) { this.dungeon = dungeon; }

    public State getState() {
        synchronized (stateLock) {
            return state;
        }
    }

    public String getLastMessage() {
        synchronized (stateLock) {
            return lastMessage;
        }
    }

    public CombatSystem getCombatSystem() { return combatSystem; }

    public void playerAction(Action action) {
        switch (action) {
            case ATTACK:
                playerAttack();
                break;
            case USE_POTION:
                playerUsePotion();
                break;
            case RUN:
                playerRun();
                break;
            case DEFEND:
                playerDefend();
                break;
        }
    }

    public void playerAttack() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerAttack called outside PLAYER_TURN");
            return;
        }
        combatSystem.playerAttack();

        if (enemy.isDefeated() || enemy.getCurrentHealth() <= 0) {
            // FIXED: Remove enemy from dungeon grid on victory
            if (dungeon != null && enemy != null) {
                int enemyX = enemy.getX();
                int enemyY = enemy.getY();
                Log.d(TAG, "Removing defeated enemy from grid at (" + enemyX + "," + enemyY + ")");
                dungeon.setTile(enemyX, enemyY, Dungeon.TILE_FLOOR);
            }
            
            setState(State.VICTORY, "Victory! " + enemy.getName() + " defeated!");
        } else {
            setState(State.ENEMY_TURN, enemy.getName() + " prepares to strike...");
        }
    }

    public void playerDefend() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerDefend called outside PLAYER_TURN");
            return;
        }
        setState(State.ENEMY_TURN, "Player braces for impact!");
    }

    public void playerRun() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerRun called outside PLAYER_TURN");
            return;
        }
        boolean fled = SecureRandomUtil.nextDouble() < 0.40;
        if (fled) {
            setState(State.FLED, "Successfully escaped!");
        } else {
            setState(State.ENEMY_TURN, "Failed to escape! " + enemy.getName() + " attacks!");
        }
    }

    public void playerUsePotion() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerUsePotion called outside PLAYER_TURN");
            return;
        }
        boolean used = player.usePotion();
        if (used) {
            setState(State.ENEMY_TURN, "Player healed! " + enemy.getName() + " approaches...");
        } else {
            synchronized (stateLock) {
                lastMessage = "Already at full health!";
            }
        }
    }

    public void enemyTurn() {
        if (getState() != State.ENEMY_TURN) {
            Log.w(TAG, "enemyTurn called outside ENEMY_TURN");
            return;
        }
        if (enemy.getCurrentHealth() <= 0 || enemy.isDefeated()) {
            if (dungeon != null && enemy != null) {
                int enemyX = enemy.getX();
                int enemyY = enemy.getY();
                dungeon.setTile(enemyX, enemyY, Dungeon.TILE_FLOOR);
            }
            setState(State.VICTORY, enemy.getName() + " has fallen!");
            return;
        }

        combatSystem.enemyAttack();

        if (player.getCurrentHealth() <= 0) {
            setState(State.DEFEAT, "You have been defeated...");
        } else {
            setState(State.PLAYER_TURN, "Your move.");
        }
    }

    public boolean isOver() {
        State s = getState();
        return s == State.VICTORY || s == State.DEFEAT || s == State.FLED;
    }

    private void setState(State newState, String message) {
        synchronized (stateLock) {
            if (state == State.VICTORY || state == State.DEFEAT || state == State.FLED) {
                Log.w(TAG, "Ignoring transition from terminal state " + state + " to " + newState);
                return;
            }
            this.state = newState;
            this.lastMessage = message;
        }
    }
}

    private static final String EXTRA_ENEMY = "enemy";
    private static final long TURN_DELAY_MS = 1000;
    private static final long RESULT_DELAY_MS = 2000;
    private static final String TAG = "BattleActivity";

        if (battleView == null) {
            Log.e(TAG, "BattleView not found in layout");
            finish();
            return;
        }
        handler = new Handler(Looper.getMainLooper());

        Intent intent = getIntent();
        Bundle extras = intent.getExtras();
        if (extras != null) {
            player = (Player) extras.getSerializable(EXTRA_PLAYER);
            enemy = (Enemy) extras.getSerializable(EXTRA_ENEMY);
            dungeon = (Dungeon) extras.getSerializable("dungeon");  // ← Get dungeon from MainActivity
        }

        if (player != null && enemy != null) {
            // Pass dungeon to Battle for floor tracking
            battle = new Battle(player, enemy, dungeon);
            battleView.setBattle(battle);
            
            battleView.loadImages(R.drawable.player_battle, R.drawable.enemy_battle);
            
            combatSystem = battle.getCombatSystem();

            if (MainActivity.getLootTableInstance() != null) {
                lootTable = MainActivity.getLootTableInstance();
                combatSystem.setLootTable(lootTable);
                Log.d(TAG, "LootTable connected to CombatSystem");
            } else {
                Log.w(TAG, "No LootTable available, loot drops disabled");
            }
            Log.d(TAG, "Battle started: " + enemy.getName() + " | Floor: " + (dungeon != null ? dungeon.getCurrentFloor() : "unknown"));
        } else {
            Log.e(TAG, "Missing player or enemy - finishing battle activity");
            setResult(RESULT_CANCELED);
            finish();
            return;
        }

        // CORRECTED: Use camelCase IDs matching your layout
        btnAttack = findViewById(R.id.btnAttack);
        btnHeal = findViewById(R.id.btnHeal);
        btnFlee = findViewById(R.id.btnFlee);
        
        if (btnAttack != null) { btnAttack.setOnClickListener(v -> playerAction(Battle.Action.ATTACK)); }
        if (btnHeal != null) { btnHeal.setOnClickListener(v -> playerAction(Battle.Action.USE_POTION)); }
        if (btnFlee != null) { btnFlee.setOnClickListener(v -> playerAction(Battle.Action.RUN)); }
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        isDestroyed = true;
        handler.removeCallbacksAndMessages(null);
    }

    public void playerAction(Battle.Action action) {
        if (battle == null || isDestroyed) {
            Log.w(TAG, "playerAction: battle is null or activity destroyed");
            return;
        }
        Log.d(TAG, "Player action: " + action);
        battle.playerAction(action);
        battleView.invalidate();
        if (isBattleOver()) {
            handler.postDelayed(this::showBattleResult, RESULT_DELAY_MS);
            return;
        }
        handler.postDelayed(this::processTurn, TURN_DELAY_MS);
    }

    private void processTurn() {
        if (battle == null || isDestroyed) { return; }

        Log.d(TAG, "Processing enemy turn");
        battle.enemyTurn();
        battleView.invalidate();
        if (isBattleOver()) {
            Log.d(TAG, "Battle over after enemy action");
            handler.postDelayed(this::showBattleResult, RESULT_DELAY_MS);
        }
    }

    private boolean isBattleOver() {
        if (battle == null) return false;
        Battle.State state = battle.getState();
        return state == Battle.State.VICTORY || state == Battle.State.DEFEAT || state == Battle.State.FLED;
    }

    private void showBattleResult() {
        Log.d(TAG, "Showing battle result");
        if (isDestroyed) return;
        if (battle == null || player == null || enemy != null) {
            Log.e(TAG, "showBattleResult: null state detected");
            setResult(RESULT_CANCELED);
            finish();
            return;
        }
        boolean battleWon = battle.getState() == Battle.State.VICTORY;
        
        if (battleWon && combatSystem != null) {
            Log.d(TAG, "Victory! Awarding rewards...");

            // FIXED: Get REAL floor from Dungeon instead of capping at 10
            int actualFloor = 1;  // Default fallback
            if (dungeon != null) {
                try {
                    actualFloor = dungeon.getCurrentFloor();
                    Log.d(TAG, "Got floor from dungeon: " + actualFloor);
                } catch (Exception e) {
                    Log.w(TAG, "Dungeon.getCurrentFloor() failed, using fallback: " + e.getMessage());
                    actualFloor = Math.min(50, (enemy.getGoldReward() / 10) + 1);
                }
            } else {
                // Fallback: estimate from gold but DON'T cap at 10 anymore
                actualFloor = Math.min(50, (enemy.getGoldReward() / 10) + 1);
                Log.d(TAG, "Dungeon null, estimated floor from gold: " + actualFloor);
            }

            Log.d(TAG, "Calling awardRewards with floor: " + actualFloor);
            combatSystem.awardRewards(player, actualFloor);
            Log.d(TAG, "Rewards granted: XP=" + player.getExperience() + ", Level=" + player.getLevel() + ", Gold=" + player.getGoldCollected());
        } else if (!battleWon) {
            Log.d(TAG, "Defeat or flee: No rewards given");
        }

        Intent resultIntent = new Intent();
        resultIntent.putExtra("enemyId", enemy.getId());
        resultIntent.putExtra("updatedPlayer", player);
        resultIntent.putExtra("battleWon", battleWon);
        
        setResult(RESULT_OK, resultIntent);
        finish();
    }
}

    public void loadImages(int playerResId, int enemyResId) {

    private void initPaints() {

        healthBarBackPaint.setStyle(Paint.Style.FILL);

        playerHealthPaint.setStyle(Paint.Style.FILL);

        enemyHealthPaint.setStyle(Paint.Style.FILL);

        textPaint = new Paint();
        textPaint.setColor(Color.WHITE);
        textPaint.setTextSize(32);
        textPaint.setTextAlign(Paint.Align.LEFT);
        textPaint.setFakeBoldText(true);
    }

    private int getHealthColor(float healthRatio) {
        healthRatio = Math.max(0f, Math.min(1f, healthRatio));
        if (healthRatio > 0.5f) {
            return Color.parseColor("#00FF00");
        } else if (healthRatio > 0.25f) {
            return Color.parseColor("#FFFF00");
        } else {
            return Color.parseColor("#FF0000");
        }
    }

    @Override

            Log.w(TAG, "Player or Enemy is null in onDraw");

            return;
        }

        int height = getHeight();
        int width = getWidth();

        // === VERTICAL BREAKDOWN ===
        // Health bars: 12.5% of BattleView
        // HERO/ENEMY labels: ~4% strip between health bars and sprites
        // Sprites: ~64% (reduced from 68.75% to make room for labels)
        // Message: remaining (~19%)
        int healthBarAreaHeight = Math.max(50, (int)(height * 0.125));
        int labelStripHeight = Math.max(30, (int)(height * 0.04));
        int spriteAreaHeight = (int)(height * 0.6875) - labelStripHeight;
        int messageAreaHeight = height - healthBarAreaHeight - labelStripHeight - spriteAreaHeight;

        int labelTop = healthBarAreaHeight;
        int imageTop = healthBarAreaHeight + labelStripHeight;
        int imageBottom = imageTop + spriteAreaHeight;

        drawHeroEnemyLabels(canvas, width, labelTop, labelStripHeight);

    private void drawEmptyState(Canvas canvas) {

        emptyPaint.setTextAlign(Paint.Align.CENTER);
        emptyPaint.setColor(Color.GRAY);

    private void drawHealthBars(Canvas canvas, int width, int healthBarAreaHeight) {
        Player player = battle.getPlayer();
        Enemy enemy = battle.getEnemy();
        if (player == null || enemy == null) return;

        // Each bar takes 50% width minus margins
        int barWidth = width / 2 - 10;
        int barHeight = Math.max(20, healthBarAreaHeight - 10);
        int centerY = healthBarAreaHeight / 2;

        // === Player health bar (left 50%) ===
        int playerBarX = 5;
        int playerBarY = centerY - barHeight / 2;
        int playerMaxHealth = player.getMaxHealth();
        int playerCurrentHealth = Math.max(0, Math.min(player.getCurrentHealth(), playerMaxHealth));
        float playerHealthRatio = (float) playerCurrentHealth / playerMaxHealth;

        canvas.drawRect(playerBarX, playerBarY, playerBarX + barWidth, playerBarY + barHeight, healthBarBackPaint);
        playerHealthPaint.setColor(getHealthColor(playerHealthRatio));
        canvas.drawRect(playerBarX, playerBarY,
                playerBarX + (int)(barWidth * playerHealthRatio),
                playerBarY + barHeight, playerHealthPaint);
        canvas.drawText("HP: " + playerCurrentHealth + "/" + playerMaxHealth,
                playerBarX + 5, centerY + 8, textPaint);

        // === Enemy health bar (right 50%) ===
        int enemyBarX = playerBarX + barWidth + 10;

        if (enemy.getCurrentHealth() > 0 && !enemy.isDefeated()) {
            int enemyMaxHealth = enemy.getMaxHealth();
            int enemyCurrentHealth = Math.max(0, Math.min(enemy.getCurrentHealth(), enemyMaxHealth));
            float enemyHealthRatio = (float) enemyCurrentHealth / enemyMaxHealth;

            canvas.drawRect(enemyBarX, playerBarY, enemyBarX + barWidth, playerBarY + barHeight, healthBarBackPaint);
            enemyHealthPaint.setColor(getHealthColor(enemyHealthRatio));
            canvas.drawRect(enemyBarX, playerBarY,

    private void handleSingleMove(Direction dir) {
        if (game == null || game.getState() != Game.GameState.PLAYING) return;
        Dungeon dungeon = game.getDungeon();
        Player player = game.getPlayer();
        if (dungeon == null || player == null) return;

        boolean encounterOccurred = false;
        if (dungeon.isWalkable(player.getX() + dir.dx, player.getY() + dir.dy)) {
            encounterOccurred = game.playerMove(dir.dx, dir.dy);
        }

        if (encounterOccurred && listener != null) {
            Enemy enemy = game.getAndClearEncounteredEnemy();
            if (enemy != null) { listener.onEncounter(enemy); }
        }

        Game.GameState state = game.getState();
        if (state != Game.GameState.PLAYING && listener != null) {
            listener.onGameOver(state);
        }
    }

            MessageDigest md = MessageDigest.getInstance("SHA-256");

            StringBuilder sb = new StringBuilder();
            for (byte b : hash) {
                sb.append(String.format("%02x", b));
            }
            return sb.toString();
        } catch (Exception e) {
            throw new IOException("Hash computation failed", e);
        }
    }

    private boolean verifyHash(File saveFile, String data) {

            return false;
        }
        try (BufferedReader reader = new BufferedReader(new FileReader(hashFile))) {
            String expected = reader.readLine();
            String actual = computeHash(data);
            boolean verified = expected != null && expected.equals(actual);
            if (!verified) {
                Log.w(TAG, "Hash mismatch for: " + saveFile.getName());
            }
            return verified;
        } catch (IOException e) {
            Log.e(TAG, "Hash verification failed: " + e.getMessage());
            return false;
        }
