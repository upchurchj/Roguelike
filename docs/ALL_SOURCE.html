package com.knightlight.game;

import android.util.Log;
import java.io.Serializable;

/**
 * Battle state manager with thread-safe state access.
 * Uses actual Player/Enemy APIs from Character inheritance chain.
 * FIXED: Proper dungeon reference for enemy cleanup on victory.
 */
public class Battle implements Serializable {
    private static final long serialVersionUID = 2L;
    private static final String TAG = "Battle";

    public enum State {
        PLAYER_TURN, ENEMY_TURN, VICTORY, DEFEAT, FLED
    }

    public enum Action {
        ATTACK, USE_POTION, RUN, DEFEND
    }

    private Player player;
    private Enemy enemy;
    private transient CombatSystem combatSystem;
    private transient Dungeon dungeon;
    private State state;
    private String lastMessage;
    private final transient Object stateLock = new Object();

    public Battle(Player player, Enemy enemy) {
        this(player, enemy, null);
    }

    public Battle(Player player, Enemy enemy, Dungeon dungeon) {
        if (player == null) {
            throw new IllegalArgumentException("Player cannot be null");
        }
        if (enemy == null) {
            throw new IllegalArgumentException("Enemy cannot be null");
        }
        this.player = player;
        this.enemy = enemy;
        this.dungeon = dungeon;
        this.combatSystem = new CombatSystem(player, enemy);
        this.state = State.PLAYER_TURN;
        this.lastMessage = "Battle started! " + enemy.getName() + " appears!";
        Log.d(TAG, "Battle init: Player Lvl " + player.getLevel()
            + " vs " + enemy.getName() + " HP:" + enemy.getCurrentHealth());
    }

    public Object readResolve() {
        if (player != null && enemy != null) {
            this.combatSystem = new CombatSystem(player, enemy);
        } else {
            Log.w(TAG, "readResolve: null player/enemy, marking battle ended");
            this.state = State.VICTORY;
        }
        return this;
    }

    public Player getPlayer() { return player; }
    public Enemy getEnemy() { return enemy; }
    public Dungeon getDungeon() { return dungeon; }
    public void setDungeon(Dungeon dungeon) { this.dungeon = dungeon; }

    public State getState() {
        synchronized (stateLock) {
            return state;
        }
    }

    public String getLastMessage() {
        synchronized (stateLock) {
            return lastMessage;
        }
    }

    public CombatSystem getCombatSystem() { return combatSystem; }

    public void playerAction(Action action) {
        switch (action) {
            case ATTACK:
                playerAttack();
                break;
            case USE_POTION:
                playerUsePotion();
                break;
            case RUN:
                playerRun();
                break;
            case DEFEND:
                playerDefend();
                break;
        }
    }

    public void playerAttack() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerAttack called outside PLAYER_TURN");
            return;
        }
        combatSystem.playerAttack();

        if (enemy.isDefeated() || enemy.getCurrentHealth() <= 0) {
            // FIXED: Remove enemy from dungeon grid on victory
            if (dungeon != null && enemy != null) {
                int enemyX = enemy.getX();
                int enemyY = enemy.getY();
                Log.d(TAG, "Removing defeated enemy from grid at (" + enemyX + "," + enemyY + ")");
                dungeon.setTile(enemyX, enemyY, Dungeon.TILE_FLOOR);
            }
            
            setState(State.VICTORY, "Victory! " + enemy.getName() + " defeated!");
        } else {
            setState(State.ENEMY_TURN, enemy.getName() + " prepares to strike...");
        }
    }

    public void playerDefend() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerDefend called outside PLAYER_TURN");
            return;
        }
        setState(State.ENEMY_TURN, "Player braces for impact!");
    }

    public void playerRun() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerRun called outside PLAYER_TURN");
            return;
        }
        boolean fled = SecureRandomUtil.nextDouble() < 0.40;
        if (fled) {
            setState(State.FLED, "Successfully escaped!");
        } else {
            setState(State.ENEMY_TURN, "Failed to escape! " + enemy.getName() + " attacks!");
        }
    }

    public void playerUsePotion() {
        if (getState() != State.PLAYER_TURN) {
            Log.w(TAG, "playerUsePotion called outside PLAYER_TURN");
            return;
        }
        boolean used = player.usePotion();
        if (used) {
            setState(State.ENEMY_TURN, "Player healed! " + enemy.getName() + " approaches...");
        } else {
            synchronized (stateLock) {
                lastMessage = "Already at full health!";
            }
        }
    }

    public void enemyTurn() {
        if (getState() != State.ENEMY_TURN) {
            Log.w(TAG, "enemyTurn called outside ENEMY_TURN");
            return;
        }
        if (enemy.getCurrentHealth() <= 0 || enemy.isDefeated()) {
            if (dungeon != null && enemy != null) {
                int enemyX = enemy.getX();
                int enemyY = enemy.getY();
                dungeon.setTile(enemyX, enemyY, Dungeon.TILE_FLOOR);
            }
            setState(State.VICTORY, enemy.getName() + " has fallen!");
            return;
        }

        combatSystem.enemyAttack();

        if (player.getCurrentHealth() <= 0) {
            setState(State.DEFEAT, "You have been defeated...");
        } else {
            setState(State.PLAYER_TURN, "Your move.");
        }
    }

    public boolean isOver() {
        State s = getState();
        return s == State.VICTORY || s == State.DEFEAT || s == State.FLED;
    }

    private void setState(State newState, String message) {
        synchronized (stateLock) {
            if (state == State.VICTORY || state == State.DEFEAT || state == State.FLED) {
                Log.w(TAG, "Ignoring transition from terminal state " + state + " to " + newState);
                return;
            }
            this.state = newState;
            this.lastMessage = message;
        }
    }
}

    private static final long RESULT_DELAY_MS = 2000;
    private static final String TAG = "BattleActivity";

    private Battle battle;
    private BattleView battleView;

    private Enemy enemy;

            return;

        Intent intent = getIntent();
        Bundle extras = intent.getExtras();

            return;

        btnDefend = findViewById(R.id.btnDefend);

        }
        if (btnDefend != null) {
            btnDefend.setOnClickListener(v -> playerAction(Battle.Action.DEFEND));
        }
        if (btnFlee != null) {
            btnFlee.setOnClickListener(v -> playerAction(Battle.Action.RUN));
        }
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        isDestroyed = true;
        handler.removeCallbacksAndMessages(null);
    }

    public void playerAction(Battle.Action action) {
        if (battle == null || isDestroyed) {
            Log.w(TAG, "playerAction: battle is null or activity destroyed");
            return;
        }
        Log.d(TAG, "Player action: " + action);
        battle.playerAction(action);
        battleView.invalidate();

        if (isBattleOver()) {
            handler.postDelayed(this::showBattleResult, RESULT_DELAY_MS);
            return;
        }

        handler.postDelayed(this::processTurn, TURN_DELAY_MS);
    }

        if (battle == null || isDestroyed) {
            return;
        }
        Log.d(TAG, "Processing enemy turn");
        battle.enemyTurn();
        battleView.invalidate();
        if (isBattleOver()) {
            Log.d(TAG, "Battle over after enemy action");
            handler.postDelayed(this::showBattleResult, RESULT_DELAY_MS);
        }
    }

    private boolean isBattleOver() {
        if (battle == null) return false;
        Battle.State state = battle.getState();
        return state == Battle.State.VICTORY || state == Battle.State.DEFEAT || state == Battle.State.FLED;
    }

     * Shows battle result and returns to MainActivity.
     * FIXED: Single call to awardRewards() - no duplication.

        if (battle == null || player == null || enemy == null) {
            Log.e(TAG, "showBattleResult: null state detected");
            setResult(RESULT_CANCELED);
            finish();
            return;
        }

        boolean battleWon = battle.getState() == Battle.State.VICTORY;

        // FIX: Award rewards exactly ONCE if battle was won
        if (battleWon && combatSystem != null) {
            Log.d(TAG, "Victory! Awarding rewards...");
            combatSystem.awardRewards(player);
            Log.d(TAG, "Rewards granted: XP=" + player.getExperience() + 
                       ", Level=" + player.getLevel() + ", Gold=" + player.getGoldCollected());
        } else if (!battleWon) {
            Log.d(TAG, "Defeat or flee: No rewards given");
        }

        Intent resultIntent = new Intent();
        resultIntent.putExtra("enemyId", enemy.getId());
        resultIntent.putExtra("updatedPlayer", player);
        resultIntent.putExtra("battleWon", battleWon);
        setResult(RESULT_OK, resultIntent);
        finish();
    }

import android.util.Log;

import java.util.Collections;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;

/**
 * Thread-safe combat resolution system.
 * FIXED: Uses Character public getters instead of fragile reflection.
 * FIXED: Central reward logic - single source of truth for XP/gold awards.
 */
public class CombatSystem {
    private static final String TAG = "CombatSystem";
    private Player player;
    private Enemy enemy;
    private final List<String> combatLog;
    private volatile String lastLogEntry;

    public CombatSystem(Player player, Enemy enemy) {
        if (player == null) {
            throw new IllegalArgumentException("Player cannot be null");
        }
        if (enemy == null) {
            throw new IllegalArgumentException("Enemy cannot be null");
        }
        this.player = player;
        this.enemy = enemy;
        this.combatLog = new CopyOnWriteArrayList<>();
        this.lastLogEntry = "";
        Log.d(TAG, "CombatSystem initialized: " + player.getName() + " vs " + enemy.getName());
    }

    public String getLastLogEntry() {
        return lastLogEntry;
    }

    public List<String> getCombatLog() {
        return Collections.unmodifiableList(new ArrayList<>(combatLog));
    }

    public Enemy getEnemy() { return enemy; }

    private void appendLog(String message) {
        if (message == null || message.isEmpty()) {
            return;
        }
        combatLog.add(message);
        lastLogEntry = message;
    }

    // ========== FIX: Use public getters instead of reflection ==========
    private int getAttack(Character c) {
        return c.getAttack();
    }

    private int getDefense(Character c) {
        return c.getArmor();
    }

    private int getCurrentHealth(Character c) {
        return c.getCurrentHealth();
    }

    private void setCurrentHealth(Character c, int health) {
        c.setCurrentHealth(health);
    }

    private int getMaxHealth(Character c) {
        return c.getMaxHealth();
    }

    public void playerAttack() {
        if (player == null || enemy == null) {
            Log.w(TAG, "playerAttack called with null player or enemy");
            return;
        }
        int playerAttack = getAttack(player);
        int enemyDefense = getDefense(enemy);
        int damage = calculateDamage(playerAttack, enemyDefense);

        boolean isCritical = SecureRandomUtil.nextDouble() < 0.15;
        if (isCritical) {
            damage *= 2;
            appendLog("CRITICAL HIT! " + damage + " damage to " + enemy.getName());
        } else {
            appendLog("Player attacks " + enemy.getName() + " for " + damage + " damage");
        }

        setCurrentHealth(enemy, getCurrentHealth(enemy) - damage);

        if (getCurrentHealth(enemy) <= 0) {
            enemy.takeDamage(9999); // Force defeat state
            appendLog(enemy.getName() + " has been defeated!");
        }
    }

    public void enemyAttack() {
        if (player == null || enemy == null) {
            Log.w(TAG, "enemyAttack called with null player or enemy");
            return;
        }
        if (getCurrentHealth(enemy) <= 0 || !enemy.isAlive()) {
            Log.w(TAG, "enemyAttack called on dead enemy");
            return;
        }

        int enemyAttack = getAttack(enemy);
        int playerDefense = getDefense(player);
        int damage = calculateDamage(enemyAttack, playerDefense);

        boolean isCritical = SecureRandomUtil.nextDouble() < 0.10;
        if (isCritical) {
            damage *= 2;
            appendLog("CRITICAL! " + enemy.getName() + " strikes for " + damage + "!");
        } else {
            appendLog(enemy.getName() + " attacks for " + damage + " damage");
        }

        setCurrentHealth(player, getCurrentHealth(player) - damage);

        if (player.getCurrentHealth() <= 0) {
            appendLog("Player has been defeated!");
        }
    }

    private int calculateDamage(int attack, int defense) {

     * Generate a single loot item based on floor and rarity threshold.
     */
    public Equipment generateLoot(int floorNumber, double rarityThreshold) {
        lock.readLock().lock();
        try {
            // Roll for loot drop
            if (random.nextDouble() > rarityThreshold) {
                return null; // No loot
            }

            // Select rarity tier based on weights
            Equipment.Rarity selectedRarity = selectRarityByWeight();
            if (selectedRarity == null) {
                return null;
            }

            List<Equipment> pool = poolsByRarity.get(selectedRarity);
            if (pool == null || pool.isEmpty()) {
                return null;
            }

            // Select random item from rarity pool
            Equipment baseItem = pool.get(random.nextInt(pool.size()));

            // Scale stats by floor: 1.0 + (floor - 1) * 0.05
            double floorScalar = 1.0 + (floorNumber - 1) * 0.05;
            int scaledATK = (int) (baseItem.getAttackBonus() * floorScalar);
            int scaledDEF = (int) (baseItem.getDefenseBonus() * floorScalar);
            int scaledHP = (int) (baseItem.getHpBonus() * floorScalar);

            return new Equipment.Builder(baseItem.getName(), baseItem.getType())
                    .rarity(baseItem.getRarity())
                    .attackBonus(scaledATK)
                    .defenseBonus(scaledDEF)
                    .hpBonus(scaledHP)
                    .description(baseItem.getDescription())
                    .build();
        } finally {
            lock.readLock().unlock();
        }
    }

     * Generate multiple loot items in a batch.
     */
    public List<Equipment> generateLootBatch(int floorNumber, double rarityThreshold, int count) {
        List<Equipment> loot = new ArrayList<>();

            Equipment item = generateLoot(floorNumber, rarityThreshold);

                .mapToDouble(Double::doubleValue)
                .sum();

        if (totalWeight <= 0) {
            return Equipment.Rarity.COMMON;

        double roll = random.nextDouble() * totalWeight;
        double cumulative = 0;

        for (Map.Entry<Equipment.Rarity, Double> entry : rarityWeights.entrySet()) {
            cumulative += entry.getValue();
            if (roll <= cumulative) {
                return entry.getKey();
            }
        }

        return Equipment.Rarity.COMMON;
