package com.knightlight.game;

import android.util.Log;
import java.io.Serializable;

/**
 * Battle state with proper deserialization handling.
 * 
 * Fixed Issues:
 * - Transient fields (combatSystem, dungeon) restored after deserialization
 */
public class Battle implements Serializable {
    private static final long serialVersionUID = 2L;
    private static final String TAG = "Battle";

    public enum State { PLAYER_TURN, ENEMY_TURN, VICTORY, DEFEAT, FLED }
    public enum Action { ATTACK, USE_POTION, RUN, DEFEND }

    private Player player;
    private Enemy enemy;
    private transient CombatSystem combatSystem;
    private transient Dungeon dungeon;
    private State state;
    private String lastMessage;
    private final transient Object stateLock = new Object();

    public Battle(Player player, Enemy enemy) { this(player, enemy, null); }

    public Battle(Player player, Enemy enemy, Dungeon dungeon) {
        if (player == null) throw new IllegalArgumentException("Player cannot be null");
        if (enemy == null) throw new IllegalArgumentException("Enemy cannot be null");
        this.player = player;
        this.enemy = enemy;
        this.dungeon = dungeon;
        this.combatSystem = new CombatSystem(player, enemy);
        this.state = State.PLAYER_TURN;
        this.lastMessage = "Battle started! " + enemy.getName() + " appears!";
        Log.d(TAG, "Battle init: Player Lvl " + player.getLevel()
            + " vs " + enemy.getName() + " HP:" + enemy.getCurrentHealth());
    }

    /**
     * FIX #3: Restore transient fields after deserialization
     */
    public Object readResolve() {
        if (player != null && enemy != null) {
            this.combatSystem = new CombatSystem(player, enemy, 
                dungeon != null ? LootTable.getInstance() : null);
            if (this.stateLock == null) {
                // Reinitialize state lock if it was null (shouldn't happen but safe)
                try {
                    java.lang.reflect.Field f = Battle.class.getDeclaredField("stateLock");
                    f.setAccessible(true);
                    f.set(this, new Object());
                } catch (Exception e) {
                    Log.w(TAG, "Failed to recreate stateLock");
                }
            }
        } else {
            Log.w(TAG, "readResolve: null player/enemy, marking battle ended");
            this.state = State.VICTORY;
        }
        return this;
    }

    public Player getPlayer() { return player; }
    public Enemy getEnemy() { return enemy; }
    public Dungeon getDungeon() { return dungeon; }
    public void setDungeon(Dungeon dungeon) { this.dungeon = dungeon; }
    public State getState() { synchronized (stateLock) { return state; } }
    public String getLastMessage() { synchronized (stateLock) { return lastMessage; } }
    public CombatSystem getCombatSystem() { return combatSystem; }

    public void playerAction(Action action) {
        switch (action) {
            case ATTACK: playerAttack(); break;
            case USE_POTION: playerUsePotion(); break;
            case RUN: playerRun(); break;
            case DEFEND: playerDefend(); break;
        }
    }

    public void playerAttack() {
        if (getState() != State.PLAYER_TURN) { Log.w(TAG, "playerAttack called outside PLAYER_TURN"); return; }
        if (combatSystem == null) {
            Log.e(TAG, "combatSystem is null - cannot attack");
            return;
        }
        combatSystem.playerAttack();
        if (enemy.isDefeated() || enemy.getCurrentHealth() <= 0) {
            enemy.setDefeated(true);
            setState(State.VICTORY, "Victory! " + enemy.getName() + " defeated!");
        } else {
            setState(State.ENEMY_TURN, enemy.getName() + " prepares to strike...");
        }
    }

    public void playerDefend() {
        if (getState() != State.PLAYER_TURN) { Log.w(TAG, "playerDefend called outside PLAYER_TURN"); return; }
        setState(State.ENEMY_TURN, "Player braces for impact!");
    }

    public void playerRun() {
        if (getState() != State.PLAYER_TURN) { Log.w(TAG, "playerRun called outside PLAYER_TURN"); return; }
        boolean fled = SecureRandomUtil.nextDouble() < 0.40;
        if (fled) {
            setState(State.FLED, "Successfully escaped!");
        } else {
            setState(State.ENEMY_TURN, "Failed to escape! " + enemy.getName() + " attacks!");
        }
    }

    public void playerUsePotion() {
        if (getState() != State.PLAYER_TURN) { Log.w(TAG, "playerUsePotion called outside PLAYER_TURN"); return; }
        boolean used = player.usePotion();
        if (used) setState(State.ENEMY_TURN, "Player healed! " + enemy.getName() + " approaches...");
        else { synchronized (stateLock) { lastMessage = "Already at full health!"; } }
    }

    public void enemyTurn() {
        if (getState() != State.ENEMY_TURN) { Log.w(TAG, "enemyTurn called outside ENEMY_TURN"); return; }
        if (enemy.getCurrentHealth() <= 0 || enemy.isDefeated()) {
            setState(State.VICTORY, enemy.getName() + " has fallen!");
            return;
        }
        if (combatSystem == null) {
            Log.e(TAG, "combatSystem is null - cannot execute enemy turn");
            setState(State.VICTORY, "Combat system error - enemy incapacitated");
            return;
        }
        combatSystem.enemyAttack();
        if (player.getCurrentHealth() <= 0) setState(State.DEFEAT, "You have been defeated...");
        else setState(State.PLAYER_TURN, "Your move.");
    }

    public void markEnemyDefeatedInDungeon() {
        if (dungeon == null || enemy == null) return;
        int enemyX = enemy.getX();
        int enemyY = enemy.getY();
        Log.d(TAG, "Removing defeated enemy from grid at (" + enemyX + "," + enemyY + ")");
        dungeon.setTile(enemyX, enemyY, Dungeon.TILE_FLOOR);
    }

    public boolean isOver() {
        State s = getState();
        return s == State.VICTORY || s == State.DEFEAT || s == State.FLED;
    }

    private void setState(State newState, String message) {
        synchronized (stateLock) {
            if (state == State.VICTORY || state == State.DEFEAT || state == State.FLED) {
                Log.w(TAG, "Ignoring transition from terminal state " + state + " to " + newState);
                return;
            }
            this.state = newState;
            this.lastMessage = message;
        }
    }
}

    public void playerAction(Battle.Action action) {
        if (isDestroyed) { android.util.Log.w(TAG, "playerAction called after destroy"); return; }
        if (battle == null) {
            android.util.Log.e(TAG, "Battle is null in playerAction");
            Toast.makeText(this, "Battle not initialized", Toast.LENGTH_SHORT).show();
            return;

        android.util.Log.d(TAG, "Player action: " + action);
        
        try {
            battle.playerAction(action);
            battleView.invalidate();
            
            if (isBattleOver()) {
                handler.postDelayed(this::showBattleResult, RESULT_DELAY_MS);
                return;
            }
            handler.postDelayed(this::processTurn, TURN_DELAY_MS);
        } catch (Exception e) {
            android.util.Log.e(TAG, "playerAction threw exception: " + e.getMessage(), e);
            Toast.makeText(this, "Action failed: " + e.getMessage(), Toast.LENGTH_SHORT).show();

            return;

        android.util.Log.d(TAG, "Processing enemy turn");
        
        try {
            battle.enemyTurn();
            battleView.invalidate();
            
            if (isBattleOver()) {
                android.util.Log.d(TAG, "Battle over after enemy action");
                handler.postDelayed(this::showBattleResult, RESULT_DELAY_MS);

            Battle.State state = battle.getState();

            android.util.Log.e(TAG, "isBattleOver threw exception: " + e.getMessage());
            return false;
        }
    }

            return;

        boolean battleWon = state == Battle.State.VICTORY;

                                break;
                            }
                        }
                    }
                    String lastLog = combatSystem.getLastLogEntry();
                    if (lastLog != null && lastLog.contains("Found")) {
                        Toast.makeText(this, "💎 " + lastLog, Toast.LENGTH_LONG).show();
                    }
                } catch (Exception e) {
                    android.util.Log.w(TAG, "Error showing loot notifications", e);
                }

                if (enemy.isBoss()) {

            }
        } else if (!battleWon) {

        }

        try {
            Intent resultIntent = new Intent();
            resultIntent.putExtra("enemyId", enemy.getId());
            resultIntent.putExtra("updatedPlayer", player);
            resultIntent.putExtra("battleWon", battleWon);
            setResult(RESULT_OK, resultIntent);
        } catch (Exception e) {
            android.util.Log.e(TAG, "Error creating result intent", e);
            setResult(RESULT_CANCELED);
        }
        finish();
    }

    private static long calculateExperienceThreshold(int level) {

        if (level >= GameBalance.MAXIMUM_PLAYER_LEVEL) {
            return Long.MAX_VALUE / 2;
        }
        double exp = BASE_EXPERIENCE * Math.pow(EXP_MULTIPLIER, level - 1);
        if (exp > Long.MAX_VALUE * 0.9) {
            return Long.MAX_VALUE;
        }
        return Math.round(exp);

    public synchronized int addExperience(int amount) {
        if (amount < 0) return 0;
        long cappedAmount = Math.min(amount, Integer.MAX_VALUE - currentExperience);
        currentExperience += cappedAmount;
        int levelsGained = 0;
        while (currentExperience >= experienceForNextLevel) {
            currentExperience -= experienceForNextLevel;
            currentLevel++;
            experienceForNextLevel = calculateExperienceThreshold(currentLevel);
            levelsGained++;
            if (currentLevel >= GameBalance.MAXIMUM_PLAYER_LEVEL) break;
        }
        Log.d(TAG, "XP added: " + cappedAmount + " | Level: " + currentLevel);
        return levelsGained;
    }
