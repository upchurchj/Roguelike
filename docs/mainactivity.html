                gamePlayer.setCurrentHealth(newHealth);
                gamePlayer.setExperience(Math.max(0, saveData.player.getExperience()));
                gamePlayer.setGoldCollected(Math.max(0, saveData.player.getGoldCollected()));
                Log.d(TAG, "Restored player stats");
            }
        }

        // Restore enemies
        if (saveData.enemies != null) {
            List currentEnemies = game.getEnemies();
            if (currentEnemies != null) {
                currentEnemies.clear();
                currentEnemies.addAll(saveData.enemies);
                Log.d(TAG, "Restored " + saveData.enemies.size() + " enemies");
            }
        }
