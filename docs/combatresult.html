package com.knightlight.game; /** * Represents the result of a combat encounter. * Holds both the outcome (VICTORY, DEFEAT, FLED, ONGOING) and any associated data. */ public class CombatResult { public enum Outcome { ONGOING, VICTORY, DEFEAT, FLED } private final Outcome outcome; private final String message; private final int experienceAwarded; private final int goldAwarded; // Private constructor — use factory methods instead private CombatResult(Outcome outcome, String message, int xp, int gold) { this.outcome = outcome; this.message = message; this.experienceAwarded = xp; this.goldAwarded = gold; } // Factory methods for common outcomes public static CombatResult ongoing() { return new CombatResult(Outcome.ONGOING, "Combat ongoing", 0, 0); } public static CombatResult victory(String message, int xp, int gold) { return new CombatResult(Outcome.VICTORY, message, xp, gold); } public static CombatResult defeat(String message) { return new CombatResult(Outcome.DEFEAT, message, 0, 0); } public static CombatResult fled(String message) { return new CombatResult(Outcome.FLED, message, 0, 0); } // Getters public Outcome getOutcome() { return outcome; } public String getMessage() { return message; } public int getExperienceAwarded() { return experienceAwarded; } public int getGoldAwarded() { return goldAwarded; } public boolean isSuccessful() { return outcome == Outcome.VICTORY; } @Override public String toString() { return outcome + (message != null ?
